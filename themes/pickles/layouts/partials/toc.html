transcript or a blockquote lands in the sidebar next to the author's own sections, which is misleading. Everything inside those two containers is dropped here before the outline is derived. The emitted markup mirrors Hugo's (
+ nested
    ), because app.js and the TOC styles key off that structure. */ -}} {{- $content := .Content -}} {{- range findRE `(?s)].*?
` $content -}} {{- $content = replace $content . "" -}} {{- end -}} {{- range findRE `(?s)].*?` $content -}} {{- $content = replace $content . "" -}} {{- end -}} {{- $items := slice -}} {{- range findRE `(?s)]*>.*?` $content -}} {{- $tag := . -}} {{- $id := "" -}} {{- with findRE `id="[^"]*"` $tag 1 -}} {{- $id = replaceRE `id="([^"]*)"` "$1" (index . 0) -}} {{- end -}} {{- $text := trim (replaceRE `<[^>]*>` "" $tag) " \n\t" -}} {{- if $id -}} {{- $items = $items | append (dict "level" (int (substr $tag 2 1)) "id" $id "text" $text) -}} {{- end -}} {{- end -}} {{- if $items -}} {{- $base := (index $items 0).level -}} {{- /* The wrapper below supplies the outermost
    , so depth starts at 1. */ -}} {{- $depth := 1 -}} {{- $out := "" -}} {{- range $i, $it := $items -}} {{- $want := add (sub $it.level $base) 1 -}} {{- if eq $i 0 -}} {{- else if gt $want $depth -}} {{- range seq (sub $want $depth) -}}{{- $out = printf "%s
      " $out -}}{{- end -}} {{- else -}} {{- $out = printf "%s" $out -}} {{- if gt $depth $want -}} {{- range seq (sub $depth $want) -}}{{- $out = printf "%s

  • %s` $out $it.id $it.text -}} {{- end -}} {{- $out = printf "%s
    • " $out -}} {{- if gt $depth 1 -}} {{- range seq (sub $depth 1) -}}{{- $out = printf "%s
" $out -}}{{- end -}} {{- end -}} {{- printf `
    %s
